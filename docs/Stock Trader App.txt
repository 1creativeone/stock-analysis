# patterns.py
import pandas as pd
import numpy as np
from pandas_ta import rsi


def double_bottom(df: pd.DataFrame, window=20, tol=0.03) -> pd.Series:
    """Return boolean Series: True on the bar where the pattern completes."""
    low = df['Low'].rolling(window).min()
    # Find two troughs with a peak in between
    trough = (df['Low'] == low)
    # Require two troughs within 2*window and similar price
    signals = pd.Series(False, index=df.index)
    for i in range(window*2, len(df)):
        recent = df.iloc[i-2*window:i]
        lows = recent['Low'][recent['Low'] == recent['Low'].min()]
        if len(lows) >= 2:
            p1, p2 = lows.index[0], lows.index[-1]
            peak = df['High'].loc[p1:p2].max()
            if abs(df['Low'].iloc[i] - lows.mean()) < tol * lows.mean():
                if df['Close'].iloc[i] > peak:          # breakout confirms
                    signals.iloc[i] = True
    return signals


def rsi_divergence(df: pd.DataFrame, period=14, lookback=30) -> pd.Series:
    df['RSI'] = rsi(df['Close'], length=period)
    signals = pd.Series(False, index=df.index)
    for i in range(lookback, len(df)):
        price = df['Low'].iloc[i-lookback:i]
        rsi_val = df['RSI'].iloc[i-lookback:i]
        if price.min() < price.iloc[:-5].min() and rsi_val.min() > rsi_val.iloc[:-5].min():
            signals.iloc[i] = True
    return signals


def breakout_52w(df: pd.DataFrame) -> pd.Series:
    high_52w = df['High'].rolling(252).max().shift(1)
    return df['Close'] > high_52w

3. backtester.py – Vectorized, No Look-Ahead
python
# backtester.py
import pandas as pd
import numpy as np


def vectorized_backtest(df: pd.DataFrame, signal: pd.Series, initial_cash=100_000):
    """
    signal: boolean Series aligned with df.index
    Returns dict with metrics + equity curve
    """
    # Align
    sig = signal.reindex(df.index).fillna(False).astype(bool)
    
    # Daily returns
    returns = df['Close'].pct_change().fillna(0)
    
    # Strategy returns (long only, exit next bar)
    strategy_returns = returns.shift(-1) * sig.shift(1)  # enter on close, exit next close
    strategy_returns = strategy_returns.fillna(0)
    
    # Equity curve
    equity = (1 + strategy_returns).cumprod() * initial_cash
    equity.iloc[0] = initial_cash
    
    # Metrics
    total_return = equity.iloc[-1] / initial_cash - 1
    cagr = (equity.iloc[-1] / initial_cash) ** (252 / len(equity)) - 1
    sharpe = np.sqrt(252) * strategy_returns.mean() / (strategy_returns.std() or np.nan)
    max_dd = (equity / equity.cummax() - 1).min()
    win_rate = (strategy_returns > 0).mean()
    num_trades = sig.sum()


    return {
        'equity': equity,
        'total_return': total_return,
        'cagr': cagr,
        'sharpe': sharpe,
        'max_dd': max_dd,
        'win_rate': win_rate,
        'num_trades': num_trades,
        'signal_dates': df.index[sig].tolist()
    }

# screener.py
import yfinance as yf
import pandas as pd
from patterns import double_bottom, rsi_divergence, breakout_52w
from backtester import vectorized_backtest
from tqdm import tqdm
import os
from datetime import datetime


SP500_TICKERS = [
    "AAPL","MSFT","GOOGL","AMZN","TSLA","META","NVDA","JPM","V","MA",
    # ... add ~500, or load from CSV
]  # Full list: https://raw.githubusercontent.com/datasets/s-and-p-500-companies/master/data/constituents_symbols.txt


DATA_DIR = "data"
RESULT_DIR = "results"
os.makedirs(DATA_DIR, exist_ok=True)
os.makedirs(RESULT_DIR, exist_ok=True)


def download(ticker, period="10y"):
    path = f"{DATA_DIR}/{ticker}.csv"
    if not os.path.exists(path):
        df = yf.download(ticker, period=period, progress=False)
        df.to_csv(path)
    else:
        df = pd.read_csv(path, index_col=0, parse_dates=True)
    return df


def run_full_scan(tickers=None, top_n=5):
    if tickers is None:
        tickers = SP500_TICKERS[:50]  # demo with 50


    results = []


    for ticker in tqdm(tickers, desc="Scanning"):
        try:
            df = download(ticker, period="10y")
            if len(df) < 500: continue


            # Run all 3 patterns
            patterns = {
                'Double Bottom': double_bottom(df),
                'RSI Divergence': rsi_divergence(df),
                '52W Breakout': breakout_52w(df)
            }


            for name, signal in patterns.items():
                if not signal.any(): continue


                bt = vectorized_backtest(df, signal)
                if bt['num_trades'] == 0: continue


                results.append({
                    'ticker': ticker,
                    'pattern': name,
                    'sharpe': bt['sharpe'],
                    'cagr': bt['cagr'],
                    'max_dd': bt['max_dd'],
                    'num_trades': bt['num_trades'],
                    'latest_signal': bt['signal_dates'][-1].strftime('%Y-%m-%d'),
                    'equity_curve': bt['equity']  # for plotting later
                })
        except Exception as e:
            print(f"Error {ticker}: {e}")


    # Rank & save
    df_res = pd.DataFrame(results)
    if len(df_res) == 0:
        print("No signals found.")
        return


    df_res = df_res.sort_values('sharpe', ascending=False)
    top = df_res.head(top_n)
    top.to_csv(f"{RESULT_DIR}/top_{top_n}.csv", index=False)
    print(f"\nTop {top_n} saved to {RESULT_DIR}/top_{top_n}.csv")
    print(top[['ticker','pattern','sharpe','cagr','latest_signal']])


    # Plot top equity curves
    import matplotlib.pyplot as plt
    plt.figure(figsize=(12,6))
    for _, row in top.iterrows():
        row['equity_curve'].plot(label=f"{row['ticker']} {row['pattern']}")
    plt.title("Top Pattern Equity Curves")
    plt.legend()
    plt.savefig(f"{RESULT_DIR}/top_equity.png")
    plt.close()


if __name__ == "__main__":
    run_full_scan(top_n=5)

import streamlit as st
import pandas as pd
import plotly.graph_objects as go
import os

# ------------------- CONFIG -------------------
st.set_page_config(
    page_title="Smart Pattern Backtester",
    page_icon="chart_with_upwards_trend",
    layout="wide",
    initial_sidebar_state="expanded"
)

def load_top_results():
    if not os.path.exists(TOP_CSV):
        return None
    df = pd.read_csv(TOP_CSV)
    df['latest_signal'] = pd.to_datetime(df['latest_signal'])
    return df


def get_equity_curve(ticker):
    path = f"{DATA_DIR}/{ticker}.csv"
    if not os.path.exists(path):
        return None
    df = pd.read_csv(path, index_col=0, parse_dates=True)
    return df['Close']


# ------------------- SIDEBAR -------------------

    st.image("assets/logo.png") if os.path.exists("assets/logo.png") else st.write("")

with col2:
    st.metric("Avg CAGR", f"{results_df['cagr'].mean():.2f}")

	Code

	Max Drawdown
	min(peak_to_trough)

	Win Rate
	% of profitable trades
	(strategy_returns > 0).mean()
